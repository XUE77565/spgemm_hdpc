const slides=[...document.querySelectorAll('.slide')];let cur=0;
const pager=document.getElementById('pager');
function show(i){cur=Math.max(0,Math.min(slides.length-1,i));slides.forEach((s,k)=>s.classList.toggle('active',k===cur));pager.textContent=(cur+1)+' / '+slides.length;}
document.addEventListener('keydown',e=>{if(e.key==='ArrowRight'||e.key===' '||e.key==='PageDown')show(cur+1);else if(e.key==='ArrowLeft'||e.key==='PageUp')show(cur-1);else if(e.key==='n'||e.key==='N')document.body.classList.toggle('show-notes');else if(e.key==='p'||e.key==='P')window.print();else if(e.key==='Home')show(0);else if(e.key==='End')show(slides.length-1);});
document.addEventListener('click',e=>{if(e.clientX>innerWidth*0.7)show(cur+1);else if(e.clientX<innerWidth*0.3)show(cur-1);});
show(0);
